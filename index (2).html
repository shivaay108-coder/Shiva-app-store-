Database setup

Connect this page to your Supabase project once. After that the store works for everyone.

  1. Create a free project at supabase.com.
  2. Open SQL Editor, paste the script below, and run it.
  3. Open Project Settings, then API. Copy the Project URL and the anon public key into SUPABASE_URL and SUPABASE_ANON_KEY at the top of the script in this file.
  4. Recommended: in Authentication, Sign In / Providers, Email, turn off Confirm email so a new account opens the store right away.

-- Shiva App Store: tables, coin rules, storage
create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  email text,
  display_name text,
  coins int not null default 0 check (coins >= 0),
  created_at timestamptz not null default now()
);
create table if not exists public.apps (
  id uuid primary key default gen_random_uuid(),
  owner uuid not null references public.profiles(id) on delete cascade,
  publisher text,
  name text not null,
  description text,
  file_path text not null,
  file_name text not null,
  file_size bigint not null default 0,
  downloads int not null default 0,
  created_at timestamptz not null default now()
);
alter table public.profiles add column if not exists display_name text;
alter table public.profiles enable row level security;
alter table public.apps enable row level security;
drop policy if exists "read own profile" on public.profiles;
create policy "read own profile" on public.profiles for select using (auth.uid() = id);
drop policy if exists "public can browse apps" on public.apps;
create policy "public can browse apps" on public.apps for select using (true);

-- 100-coin welcome bonus on first signup
create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, email, display_name, coins)
  values (new.id, new.email, coalesce(new.raw_user_meta_data->>'user_name', new.raw_user_meta_data->>'full_name', split_part(new.email, '@', 1)), 100)
  on conflict (id) do nothing;
  return new;
end $$;
drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created after insert on auth.users
  for each row execute function public.handle_new_user();

-- publishing deducts 10 coins (atomic, server-side)
create or replace function public.publish_app(p_name text, p_description text, p_file_path text, p_file_name text, p_file_size bigint)
returns public.apps language plpgsql security definer set search_path = public as $$
declare r public.apps; who text;
begin
  update public.profiles set coins = coins - 10
    where id = auth.uid() and coins >= 10 returning coalesce(display_name, split_part(email, '@', 1)) into who;
  if not found then raise exception 'Not enough coins. Publishing costs 10 coins.'; end if;
  insert into public.apps (owner, publisher, name, description, file_path, file_name, file_size)
    values (auth.uid(), who, p_name, p_description, p_file_path, p_file_name, p_file_size) returning * into r;
  return r;
end $$;
revoke execute on function public.publish_app from public, anon;
grant execute on function public.publish_app to authenticated;

create or replace function public.increment_download(p_id uuid) returns void
language sql security definer set search_path = public as $$
  update public.apps set downloads = downloads + 1 where id = p_id;
$$;

-- file storage (public bucket, users write only to their own folder)
insert into storage.buckets (id, name, public, file_size_limit)
  values ('apps', 'apps', true, 52428800) on conflict (id) do nothing;
drop policy if exists "upload to own folder" on storage.objects;
create policy "upload to own folder" on storage.objects for insert to authenticated
  with check (bucket_id = 'apps' and (storage.foldername(name))[1] = auth.uid()::text);
drop policy if exists "remove own files" on storage.objects;
create policy "remove own files" on storage.objects for delete to authenticated
  using (bucket_id = 'apps' and (storage.foldername(name))[1] = auth.uid()::text);
